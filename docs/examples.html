Examples by application

Each example below includes its Python source code, a download link and a link to module documentation. The examples come directly from src/ase2sprkkr/examples/ so the displayed code follows the source.

For a first calculation, follow First calculation: aluminium SCF and DOS. They demonstrate API use rather than numerically converged reference calculations. Some require additional programs or optional Python packages. Review numerical parameters before using them in production. For scientific assumptions and parameter convergence, consult the SPR-KKR manual.

The command-line interface can list, show and copy the same examples:

ase2sprkkr example
ase2sprkkr example --regex 'Fe|DOS'
ase2sprkkr example 1
ase2sprkkr example 1 --copy ./my-example

 1"""
 2Ni Doped SrTiO3 as calculated by the CPA method. This example demonstrate partial occupations.
 3"""
 4
 5
 6def main():
 7    from ase.lattice.tetragonal import SimpleTetragonalFactory
 8    from ase2sprkkr.sprkkr.calculator import SPRKKR
 9    from ase2sprkkr.sprkkr.sprkkr_atoms import SPRKKRAtoms
10
11    # Define a Perovskite Factory class
12    class PerovskiteFactory(SimpleTetragonalFactory):
13        bravais_basis = [[0, 0, 0.0], [0.5, 0.5, 0.5], [0.0, 0.5, 0.5], [0.5, 0.5, 0], [0.5, 0.0, 0.5]]

17

29    calculator.input_parameters.set(NE=32)
30    calculator.input_parameters.set(NITER=20)
31    calculator.input_parameters.SCF.MIX = 0.01
32    calculator.input_parameters.ENERGY.ImE = 0.0
33    calculator.input_parameters.ENERGY.GRID = 5
34    out = calculator.calculate()

A05_Al_EOS

Total energy SCF calculations as a function of Volume and corresponding equation of state fiting.

  • Module documentation

  • Download Al.py

  • Copy the example: ase2sprkkr example 5 --copy ./my-example

Al.py
 1"""Total energy SCF calculations as a function of Volume and corresponding
 2equation of state fiting."""
 3
 4def main():
 5
 6    from ase.build import bulk
 7    from ase2sprkkr.sprkkr.calculator import SPRKKR
 8    from ase.io.trajectory import Trajectory
 9    import numpy as np
10    from ase.io import read
11    from ase.units import kJ
12    from ase.eos import EquationOfState
13
14
15    al = bulk("Al", "fcc", a=4.0)
16    cell = al.get_cell()
17    traj = Trajectory("Al.traj", "w")
18
19    # run calculation
20    calc = SPRKKR(atoms=al, mpi=True)
21    calc.input_parameters.set(NL=3)
22    calc.input_parameters.set(NE=32)
23    calc.input_parameters.SCF.MIX = 0.20
24    calc.input_parameters.ENERGY.ImE = 0.0
25    calc.input_parameters.ENERGY.GRID = 5
26    calc.input_parameters.SCF.FULLPOT = True
27    calc.input_parameters.MODE.MODE = "SREL"
28
29    al.set_calculator(calc)
30
31    for x in np.linspace(0.80, 1.20, 10):
32        al.set_cell(cell * x, scale_atoms=True)
33        al.get_potential_energy()
34        traj.write(al)
35
36    # Now lets plot the result and fit EOS
37    configs = read("Al.traj@0:")  # read 5 configurations
38    # Extract volumes and energies:
39    volumes = [al.get_volume() for al in configs]
40    energies = [al.get_potential_energy() for al in configs]
41    eos = EquationOfState(volumes, energies)
42    v0, e0, B = eos.fit()
43    print(B / kJ * 1.0e24, "GPa")
44    eos.plot("Al-eos.png")
45
46
47# Just run the script only when directly called from command line

49if __name__ == "__main__":
50    main()

A06_Cu_MSSPEC

Simple SCF calculations for fcc Cu in combination with msspec code to generate photoelectron diffraction.

  • Module documentation

  • Download Cu.py

  • Copy the example: ase2sprkkr example 6 --copy ./my-example

Cu.py
 1#!/usr/bin/env python
 2"""
 3Simple SCF calculations for fcc Cu in combination with msspec code to generate
 4photoelectron diffraction.
 5
 6Call with ``msspec`` and/or ``sprkkr`` commandline argument.
 7"""
 8
 9import glob
10import sys
11from ase2sprkkr.sprkkr.calculator import SPRKKR
12from ase.build import bulk
13
14
15def main(args):
16    # Create a copper cell
17    Cu = bulk("Cu")
18
19    # ########## SPRKKR part
20    if "sprkkr" in args:
21        # create a SPRKKR calculator
22        calc = SPRKKR(atoms=Cu, mpi=True)
23
24        # launch kkrscf
25        calc.input_parameters.set(NL=3)
26        calc.input_parameters.SCF.MIX = 0.20
27        calc.input_parameters.ENERGY.ImE = 0.0
28        calc.input_parameters.ENERGY.GRID = [5, 3]
29        calc.input_parameters.set(NE=32)
30        out_scf = calc.calculate()
31        #
32        # EXPORT POTENTIAL FOR PHAGEN
33        #
34
35        # Now use the calculator with the newly converged potential
36        calc = out_scf.calculator
37        calc.input_parameters = "PHAGEN"
38        calc.input_parameters.SCF.MIX = 0.20
39        calc.input_parameters.ENERGY.ImE = 0.0
40        calc.input_parameters.ENERGY.GRID = [5, 3]
41        calc.input_parameters.set(NE=32)
42        calc.calculate(potential=out_scf.potential_filename)
43
44    # ######### MsSpec part
45    if "msspec" in args:
46        from msspec.calculator import MSSPEC
47        from msspec.utils import get_atom_index, hemispherical_cluster, SPRKKRPotential
48
49        pot = SPRKKRPotential(Cu, "Cu_scf.pot_new", *glob.glob("*PHAGEN.pot"))
50
51        nplanes = 3
52        cluster = hemispherical_cluster(Cu, planes=nplanes, emitter_plane=nplanes - 1)
53        cluster.absorber = get_atom_index(cluster, 0, 0, 0)
54
55        calc = MSSPEC(folder="calc")
56        calc.set_atoms(cluster)
57        calc.tmatrix_parameters.potential = pot
58
59        data = calc.get_theta_scan(level="2p3/2")
60        data.view()
61
62    if len(args) <= 1:
63        print("Please specify either 'sprkkr', 'msspec' keywords or both of them on the command line")
64
65
66# Just run the script only when directly called from command line
67
68if __name__ == "__main__":
69    main(sys.argv)

A07_SrTiO3_MSSPEC

SCF calculations for SrTiO3 in combination with msspec code to generate photoelectron diffraction.

  • Module documentation

  • Download STO.py

  • Copy the example: ase2sprkkr example 7 --copy ./my-example

STO.py
  1"""
  2SCF calculations for  SrTiO3 in combination with msspec code to generate
  3photoelectron diffraction.
  4
  5Call with ``msspec`` and/or ``sprkkr`` commandline argument.
  6"""
  7
  8
  9def main():
 10
 11    from ase.lattice.tetragonal import SimpleTetragonalFactory
 12    from ase.visualize import view
 13    from ase2sprkkr import SPRKKR
 14    import numpy as np
 15    import sys
 16    import glob
 17
 18    args = sys.argv
 19
 20    # Define a Perovskite Factory class
 21    class PerovskiteFactory(SimpleTetragonalFactory):
 22        bravais_basis = [[0, 0, 0.0], [0.5, 0.5, 0.5], [0.0, 0.5, 0.5], [0.5, 0.5, 0], [0.5, 0.0, 0.5]]
 23        element_basis = (0, 1, 2, 2, 2)
 24
 25    Perovskite = PerovskiteFactory()
 26
 27    # Generate the base STO cell
 28    a0 = 3.905
 29    STO = Perovskite(("Sr", "Ti", "O"), latticeconstant={"a": a0, "c/a": 1.0}, size=(1, 1, 1))
 30    if "view" in args:
 31        view(STO)
 32
 33    # ########## SPRKKR part
 34    if "sprkkr" in args:
 35        calc = SPRKKR(atoms=STO, mpi=["mpirun", "-np", "16"])
 36        calc.input_parameters.set(NL=3)
 37        calc.input_parameters.SCF.MIX = 0.01
 38        calc.input_parameters.SCF.NITER = 1000
 39        calc.input_parameters.CONTROL.add("NOSYM", True)
 40
 41        out_scf = calc.calculate()
 42        calc.input_parameters = "PHAGEN"
 43        calc.calculate(potential=out_scf.potential_filename)
 44
 45    # ######### MsSpec part
 46    if "msspec" in args:
 47        from msspec.calculator import MSSPEC
 48        from msspec.utils import SPRKKRPotential, hemispherical_cluster, get_atom_index
 49
 50        ###########################################################################
 51        # Build the SrTiO3 cluster                                                #
 52        ###########################################################################
 53        # We start by loading the potential since it updates info in the STO cell
 54        pot = SPRKKRPotential(STO, "SrTiO3_scf.pot_new", *glob.glob("*PHAGEN.pot"))
 55
 56        # tag each atom in the STO object to easily set the emitter in the
 57        # hemispherical_cluster function
 58        [atom.set("tag", ("Sr", "Ti", "O").index(atom.symbol)) for atom in STO]
 59
 60        # Create a hemispherical cluster centered on a Ti emitter from the
 61        # STO primitive cell used in the SPRKKR step.
 62        # For this example we use 4 planes and the Ti emitter (tag #1) is
 63        # located in the 2nd plane (numbering starts at 0)
 64        cluster = hemispherical_cluster(STO, planes=4, emitter_plane=1, emitter_tag=1)
 65
 66        # The created cluster is centered on the emitter atom, so defining
 67        # the absorber attribute is straightforward:
 68        cluster.absorber = get_atom_index(cluster, 0, 0, 0)
 69
 70        ###########################################################################
 71        # Set up the PhotoElectron Diffraction calculator                         #
 72        ###########################################################################
 73        # Create the calculator
 74        calc = MSSPEC(spectroscopy="PED", algorithm="expansion", folder="PED")
 75
 76        # minimalistic set of parameter for XPD scan
 77        calc.set_atoms(cluster)
 78        calc.calculation_parameters.scattering_order = 2
 79        # We need to impose a maximum number of tl to use because there is still
 80        # a memory bug that I need to investigate. Anyway, usually 25 is not that
 81        # bad for the kind of atoms and the photon energy of lab sources...
 82        calc.tmatrix_parameters.lmax_mode = "imposed"
 83        calc.tmatrix_parameters.lmaxt = 25
 84
 85        data = None
 86        for source_energy in np.arange(500.0, 1501.0, 100.0):
 87            calc.source_parameters.energy = source_energy
 88            # Run a polar scan with the default MuffinTin potential for Ti(2p3/2)
 89            calc.tmatrix_parameters.potential = "muffin_tin"
 90            data = calc.get_theta_scan(level="2p3/2", data=data)
 91
 92            # Now we use the previously generated SPRKKR potential and run the same
 93            # calculation
 94            calc.tmatrix_parameters.potential = pot
 95            data = calc.get_theta_scan(level="2p3/2", data=data)
 96
 97            # To better see the differences, plot the normalized signal on the
 98            # same graph
 99            # add a new dataset for storing normalized values
100            dset = data.add_dset(f"comparison at {source_energy} eV")
101            # make a copy of previous scans values
102            theta, muffintin_cs, sprkkr_cs = (
103                data[-3].theta.copy(),
104                data[-3].cross_section.copy(),
105                data[-2].cross_section.copy(),
106            )
107            # divide by the max
108            sprkkr_cs /= sprkkr_cs.max()
109            muffintin_cs /= muffintin_cs.max()
110            # add a new dataset with those values
111            dset.add_columns(theta=theta, sprkkr=sprkkr_cs, muffintin=muffintin_cs)
112            # add a view for this dataset
113            dview = dset.add_view(
114                "Comparison",
115                title=(
116                    "Comparison of XPD polar scans for "
117                    r"Ti(2p3/2) at $h\nu$ = {:.0f} eV".format(source_energy)
118                ),
119                xlabel=r"$\Theta (\degree$)",
120                ylabel="Normalized Signal (a.u.)",
121            )
122            dview.select("theta", "sprkkr", legend="With SPRKKR potential")
123            dview.select("theta", "muffintin", legend="With internal MT potential")
124            dview.set_plot_options(autoscale=True)
125
126        # Pop up the final result
127        data.view()
128
129    if len(args) <= 1:
130        print("Please specify either 'sprkkr', 'msspec' keywords or both of them on the command line")
131
132
133# Just run the script only when directly called from command line
134
135if __name__ == "__main__":
136    main()

  • Module documentation

  • Download bcc_Fe_dlm.py

  • Copy the example: ase2sprkkr example 10 --copy ./my-example

bcc_Fe_dlm.py
 1r"""
 2Script demonstrating how to set up a disordered local moment (DLM) calculation using bcc Fe as the test case.
 3
 4Such calculations are used to describe the paramagnetic state of materials,
 5where local magnetic moments on atoms persist, but where the net magnetisation
 6of the material is zero on account of thermally induced spin fluctuations. The
 7CPA is used to average over possible orientations of magnetic moment. Above the Curie
 8temperature, all orientations of magnetic moment are equally probably, and the
 9probability distribution of orientations of magnetic moments is uniform on the sphere.
10In the absence of spin-orbit coupling, the CPA condition for a full average (using
11an angular integral) over the sphere with uniform probability distribution can be shown
12to be mathematically equivalent to the CPA condition for a 50:50 Ising 'alloy' of 'up'
13and 'down' magnetic moments. (Note that this equality does not strictly hold in the
14case where spin-orbit effects are present, as in a fully relativistic calculation.)
15
16A DLM calculation should be iterated to self-consistency to check whether the local
17moments 'collapse' (as frequently happens for elements like Cr, Ni) or whether they
18are self-sustaining (as is typical for Fe, Co).
19
20Once you have run this script and iterated to self-consistency, if you run
21`tail Fe.pot_new`
22you should find that, although the net magnetisation of the calculation is zero,
23the two Fe atoms have equal and opposite magnetic moments, each of magnitude
24approximately 2.16 \mu_B. This confirms that the local moments are self-sustaining,
25and that bcc Fe is a 'good' local moment system.
26
27Some appropriate references for the disordered local moment (DLM) picture are:
28 - A. J. Pindor et al., J. Phys. F: Met. Phys. 13, 979 (1983)
29 - J. B. Staunton et al., J. Magn. Magn. Mater. 45, 15-22 (1984)
30 - B. L. Gyorffy et al., J. Phys. F: Met. Phys. 15, 1337 (1985)

32Author: Christopher D. Woodgate, University of Bristol, 2025

34Email: christopher.woodgate@bristol.ac.uk
35"""

38def main():

40    from ase.build import bulk
41    from ase2sprkkr.sprkkr.calculator import SPRKKR
42    from ase2sprkkr.sprkkr.sprkkr_atoms import SPRKKRAtoms
43
44    # BCC Fe has a_lat = 2.87 \AA
45    a_lat = 2.87
46    atom = bulk("Fe", "bcc", a=a_lat)
47
48    # Now set up the 'alloy'
49    atoms = SPRKKRAtoms.promote_ase_atoms(atom)
50
51    # We cannot use a Python dictionary here, as we need an alloy of two elements
52    # with the same chemical symbol and this behaviour is not permitted in a Python dictionary
53    # Instead, we use a list.
54    atoms.sites[0].occupation.set([("Fe", 0.5), ("Fe", 0.5)])
55
56    # Now we set up the ASE calculator

58

60    # NOTE: These are in units of \mu_B
61    calculator.input_parameters.SCF.MSPIN = [2.3, -2.3]
62

68    calculator.input_parameters.set(NE=32)  # 32-point semicircular energy contour
69    calculator.input_parameters.ENERGY.GRID = 5  # Semicircular energy contour
70    calculator.input_parameters.SCF.MIX = 0.05  # Gentle mixing paramter to ensure convergence
71    calculator.input_parameters.SCF.TOL = 1e-7  # Tight tolerance on SCF cycle
72    calculator.input_parameters.SCF.NITER = 1000  # Allow for lots of iterations if needed
73    calculator.input_parameters.ENERGY.ImE = 0.00  # Distance of closest approach to real axis
74    calculator.input_parameters.CPA.NITER = 100  # Allow for a lot of CPA iterations if needed
75    calculator.input_parameters.CPA.TOL = 1e-8  # Tight CPA tol to go with tight SCF tol
76    calculator.input_parameters.SCF.USEVMATT = True  # I like to use the Mattheiss construction for the potential
77
78    # Run the calculator
79    out = calculator.calculate()
80
81    print(out.energy)
82    print(out.last_iteration["energy"]["EF"].to_dict())
83    print(out.iterations[-1]["error"]())
84    print(out.last_iteration["moment"].to_dict())
