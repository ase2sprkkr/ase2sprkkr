def _mpi_command(launch):
    """Materialize portable MPI intent for this machine."""
    mpi = launch.mpi
    command = resolve_mpi_command(mpi)
    if not command:
        if mpi.allows_serial_fallback:
            if mpi.warning:
                warnings.warn(
                    "MPI autodetection did not find an MPI launcher; "
                    "the calculation will run without MPI.",
                    MpiNotFoundWarning,
                    stacklevel=3,
                )
            return None
        raise FileNotFoundError(
            errno.ENOENT,
            "MPI was requested but no MPI launcher was found",
        )
    return command

def materialize_local_spec(spec: ExecutionSpec) -> ExecutionSpec:
    """Choose serial/MPI argv immediately before local process creation."""
    if not spec.launch or spec.mpi_argv is None:
        return spec

    launcher = _mpi_command(spec.launch)
    if launcher is None:
        # ``auto`` may fall back to the serial command on this machine.
        return spec

    return replace(
        spec,
        argv=(*launcher, *spec.mpi_argv),
        stdin=None,
    )

def _path(value, directory):
    if value is None:
        return None
    value = getattr(value, "name", value)
    if isinstance(value, int):
        return None
    try:
        path = os.fspath(value)
    except TypeError:
        return None
    if not os.path.isabs(path):
        path = os.path.join(directory, path)
    return os.path.abspath(path)

def _pid_running(pid):
    try:
        os.kill(pid, 0)
    except ProcessLookupError:
        return False
    except PermissionError:
        return True
    except OSError as exc:
        if exc.errno == errno.ESRCH:
            return False
        if exc.errno == errno.EPERM:
            return True
        raise
    return True

    """One local subprocess execution.

    The runner owns process lifecycle, cancellation, output-file writing and
    delivery of raw stdout/stderr to listeners. The output reader only parses
    streams and builds the TaskResult.

    def _emit_output(self, data, kind):
        if not data:
            return

        text = data.decode("utf8")
        if self.reader.print_output:
            print(text, end="", file=sys.stderr if kind == "err" else sys.stdout)
        self._output_observers.emit(text, kind)

    async def _pump(self, source, target, kind):
        try:
            while data := await source.read(65536):
                self._write_output(data, kind)
                self._emit_output(data, kind)
                target.feed_data(data)
        finally:
            target.feed_eof()

    async def _stop_process(self):
        process = self._process
        if process is None or process.returncode is not None:
            return

        try:
            process.terminate()
        except ProcessLookupError:
            return

        try:
            await asyncio.wait_for(process.wait(), 0.1)
        except asyncio.TimeoutError:
            if process.returncode is not None:
                return
            try:
                process.kill()
            except ProcessLookupError:
                return
            await process.wait()

    def _ensure_stop_task(self):
        if self._process is None or self._process.returncode is not None:
            return None
        if self._stop_task is None or self._stop_task.done():
            self._stop_task = asyncio.create_task(self._stop_process())
        return self._stop_task

    def cancel(self):
        """Request cancellation from any thread."""
        self._cancel_requested.set()

        loop = self._loop
        if loop is None:
            if self._state == RunState.NEW:
                self._state = RunState.CANCELLED
            return

        try:
            loop.call_soon_threadsafe(self._schedule_stop)
        except RuntimeError:
            # The owning event loop has already stopped.
            pass

    def _close_output_files(self):
        closed = set()
        for file in (self.spec.output_file, self.spec.error_file):
            if file is None or id(file) in closed:
                continue
            closed.add(id(file))
            if hasattr(file, "close"):
                file.close()

    async def _execute(self):
        self._loop = asyncio.get_running_loop()
        execution_finished = False

        if self._cancel_requested.is_set():
            self._state = RunState.CANCELLED
            self._started.set()
            raise RuntimeError("Run was cancelled before process start")

        try:
            self._process = await asyncio.create_subprocess_exec(
                *self.spec.argv,
                stdin=self.spec.stdin,
                stdout=asyncio.subprocess.PIPE,
                stderr=asyncio.subprocess.PIPE,
                cwd=self.spec.directory,
            )
            self._pid = self._process.pid
            self._state = RunState.RUNNING
            self._started.set()

            if self._cancel_requested.is_set():
                stop_task = self._ensure_stop_task()
                if stop_task is not None:
                    await stop_task

            stdout = asyncio.StreamReader()
            stderr = asyncio.StreamReader()
            parse_task = asyncio.create_task(
                self.reader.parse_streams(stdout, stderr, self.result)
            )
            wait_task = asyncio.create_task(self._process.wait())
            stdout_task = asyncio.create_task(
                self._pump(self._process.stdout, stdout, "out")
            )
            stderr_task = asyncio.create_task(
                self._pump(self._process.stderr, stderr, "err")
            )
            tasks = (parse_task, wait_task, stdout_task, stderr_task)

            try:
                return_code, _, _ = await asyncio.gather(
                    wait_task, stdout_task, stderr_task
                )
                self._state = (
                    RunState.CANCELLED
                    if self._cancel_requested.is_set()
                    else RunState.FINISHED
                    if return_code == 0
                    else RunState.FAILED
                )
                execution_finished = True
                output, error = await parse_task
                return self.reader.complete_result(
                    self.result, output, error, return_code
                )
            finally:
                for task in tasks:
                    if not task.done():
                        task.cancel()
                await asyncio.gather(*tasks, return_exceptions=True)
        except BaseException:
            self._started.set()
            if not execution_finished:
                self._state = (
                    RunState.CANCELLED
                    if self._cancel_requested.is_set()
                    else RunState.FAILED
                )
                if self._process is not None and self._process.returncode is None:
                    await self._stop_process()
            raise
        finally:
            if self._stop_task is not None:
                await asyncio.gather(self._stop_task, return_exceptions=True)
            self._close_output_files()
            self._process = None
            self._loop = None

    def _thread_main(self):
        try:
            asyncio.run(self._execute())
        except BaseException as exc:
            self._exception = exc
        finally:
            try:
                if self.callback:
                    self.callback(self.result)
            except BaseException as exc:
                self._callback_exception = exc
            self._done.set()

    def submit(self):
        """Start the run and return once its persistent process id is known."""
        with self._lock:
            if self._state == RunState.CANCELLED and self._thread is None:
                raise RuntimeError("Run was cancelled before submission")
            if self._state != RunState.NEW:
                raise RuntimeError(f"Run is already {self._state.value}")

            self._state = RunState.QUEUED
            self._thread = Thread(target=self._thread_main, daemon=True)
            self._thread.start()

        self._started.wait()
        if self._pid is None:
            self._done.wait()
            if self._exception is not None:
                raise SubmissionNotStarted(str(self._exception)) from self._exception

    def wait(self):
        """Wait for a submitted run and return its TaskResult."""
        if self._state == RunState.NEW:
            raise RuntimeError("Run has not been submitted")
        if self._state == RunState.CANCELLED and self._thread is None:
            raise RuntimeError("Run was cancelled before submission")

        self._done.wait()
        if self._exception is not None:
            raise self._exception
        if self._callback_exception is not None:
            raise self._callback_exception
        return self.result

    def run(self):
        """Submit, if needed, and wait for completion."""
        if self._state == RunState.NEW:
            self.submit()
        return self.wait()

    async def run_async(self):
        """Async wrapper around submit/wait."""
        if self._state == RunState.NEW:
            self.submit()
        loop = asyncio.get_running_loop()
        return await loop.run_in_executor(None, self.wait)

    def status(self, *references: RunReference):
        from .local_persistent import detached_local_status

        for reference in references:
            if reference.executor != self.name:
                raise ValueError(
                    f"Run reference belongs to {reference.executor!r}, not {self.name!r}"
                )

        statuses = []
        for reference in references:
            detached = detached_local_status(reference)
            if detached is not None:
                statuses.append(detached)
                continue

            try:
                pid = int(reference.native_id)
            except (TypeError, ValueError):
                state = RunState.UNKNOWN
            else:
                # Legacy records used a process PID as their only controller.
                # Keep status compatibility, but such runs cannot be attached.
                state = RunState.RUNNING if _pid_running(pid) else RunState.UNKNOWN
            statuses.append(RunStatus(reference, state))
        return tuple(statuses)
